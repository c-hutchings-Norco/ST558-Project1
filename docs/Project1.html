Part 1: Functions

Create a function to get the specified variables using the Census api. We are looking to get the variables along with their values.

#Create a function to get all the variables named funcation variable list (fun_var_list)
fun_var_list <- function(year) {
  
  # Exclude any years outside of the following    
  if (!year %in% c(2021, 2022, 2023, 2024)) {
    stop("Year must be 2021, 2022, 2023, or 2024.")
  }
  
  # Use the API to get the data, use paste to add the year as a variable determined by the function    
  response <- GET(
    paste0("https://api.census.gov/data/",
           year,
           "/acs/acs1/pums/variables.json"))
  
  # Parse the data from JSON  
  parse_var <- fromJSON(rawToChar(response$content))
  
  # Pull out the variables required.  Choose the names depending on the year requested because they change for certain years.  
  
  if (year %in% c(2021, 2022)) {
    required_vars <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP", "JWMNP", "PWGTP", "FER", "HHL", "SCH","SCHL", "SEX", "ST", "REGION", "DIVISION")
    
  } else {
    required_vars <- c("AGEP", "GASP", "GRPIP", "JWAP", "JWDP","JWMNP", "PWGTP", "FER", "HHL", "SCH", "SCHL", "SEX", "STATE", "REGION", "DIVISION")
  }
  
  #create a list that includes only the required variables.    
  var_list <- parse_var$variables[required_vars]
  
  #return the list  
  return(var_list)
}

#Helper function to create a tibble from parsed data
helper <- function(parsed_data){
  new_tibble <- as_tibble(parsed_data[-1, ], 
                          .name_repair='minimal')
  names(new_tibble) <- parsed_data[1, ]
  return(new_tibble)

Helper Function to find Midpoint Time

Create the time helper function that finds the midpoint of the time in the given time interval. There are several different variables available to test the time helper function on. One of the variables JWMNP is essentially a null value variable as the only value is they did not travel to work that day.

#loading the all variables object so the items can be used to identify the labels for the variables.
all_var_list <- readRDS("data/variable_list.RDS")


#Create the helper time function
helper_time <- function(code, var_name, year_metadata){
  #code 0 as NA based on the item for the value
  if(is.na(code) || code == "0"){
    return(NA)
  }
  #Pull the label from the variable using its value
  label <-
    year_metadata[[var_name]]$values$item[[as.character(code)]]
  
  if(is.null(label)){
    return(NA)
  }
  #Format the time into a usable format by removing to and fixing p.m.
  times <- strsplit(label, " to ")[[1]]
  times <- gsub("\\.","", times)
  
  start <- parse_date_time(times[1], "I:M p")
  end   <- parse_date_time(times[2], "I:M p")
  
  #Finding the midpoint of the two times in the label
  midpoint <- start + (end - start)/2
  
  #Return the midpoint in hour and minute format
  return(hm(format(midpoint, "%H:%M")))
}

# Create a factor helper function
helper_factor <- function(x, labels){
  factor(
    x,
    levels = names(labels),
    labels = unlist(labels)
  )
}
